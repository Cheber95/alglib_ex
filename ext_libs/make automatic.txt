jdeps --ignore-missing-deps --generate-module-info javafx-sdk-17.0.10/lib/jars javafx-sdk-17.0.10/lib/javafx-swt.jar
javac --patch-module javafx.swt=javafx-sdk-17.0.10/lib/javafx-swt.jar javafx-sdk-17.0.10/lib/jars/javafx.swt/module-info.java
jar uf javafx-sdk-17.0.10/lib/javafx-swt.jar -C javafx-sdk-17.0.10/lib/ module-info.class

jdeps --ignore-missing-deps --generate-module-info jars jars/jtransforms-2.4.0.jar

javac --patch-module jtransforms=jars/jtransforms/jtransforms-2.4.0.jar jars/module-info.java
 -- убедиться, что появился module-info.class. Родить обратно .jar 
jar uf jars/jtransforms-2.4.0.jar -C jars module-info.class
